"""FastAPI binding for AIML-01 Fraud Detection Platform.

Run:
    python -m uvicorn backend.app.main:create_app --factory

Business logic remains in service.py, validation.py and db.py.
Authentication is optional for local development and enabled through API_KEY.
"""

from fastapi.exceptions import RequestValidationError

from backend.app.config import Settings

API = "/api/v1"

CSV_TYPES = {
    "text/csv",
    "application/csv",
    "text/plain",
    "application/vnd.ms-excel",
}

PUBLIC_PATHS = {
    f"{API}/health",
    f"{API}/ready",

def create_app(settings: Optional[Settings] = None) -> FastAPI:
    """Create and configure the FastAPI application."""

    settings = settings or Settings.from_env()

    logger = setup_logging(settings.log_level)

    # Initialize persistence and model services.
    repo = Repository(settings.database_path)
    repo.init_schema()

    model = ModelService(settings.model_dir)
    model.load()

    service = FraudService(settings, repo, model)

    limiters = {
        "predict": RateLimiter(settings.rate_predict_per_min),
        "batch": RateLimiter(settings.rate_batch_per_min),
        "read": RateLimiter(settings.rate_read_per_min),
    }

    docs_url = None if settings.is_production else "/docs"
    openapi_url = (
        None if settings.is_production else "/openapi.json"
    )

    app = FastAPI(
        title="AIML-01 Fraud Detection API",
        version="0.3.0",
        docs_url=docs_url,
        redoc_url=None,
        openapi_url=openapi_url,
    )

    app.state.service = service
    app.state.settings = settings

    app.add_middleware(
        CORSMiddleware,
        allow_origins=list(settings.cors_origins),
        allow_credentials=False,
        allow_methods=["GET", "POST", "PUT", "PATCH"],
        allow_headers=["Authorization", "Content-Type", "X-API-Key"],
        expose_headers=["X-Request-ID", "Retry-After"],
    )

    # ---------------------------------------------------------
    # Request identity and rate limiting
    # ---------------------------------------------------------

    def request_id(request: Request) -> str:
        return getattr(request.state, "request_id", "")

    def client_key(request: Request) -> str:
        """Resolve client IP.

        TRUST_PROXY must only be enabled when the application is
        behind a trusted proxy that strips client-supplied forwarding
        headers and writes its own X-Forwarded-For header.
        """

        if settings.trust_proxy:
            forwarded = request.headers.get(
                "x-forwarded-for", ""
            )

            if forwarded:
                return forwarded.split(",")[0].strip()[:64]

        if request.client:
            return request.client.host

        return "unknown"

    def limit(bucket: str):
        def dependency(request: Request) -> None:
            limiter = limiters[bucket]

            allowed, retry_after = limiter.check(
                client_key(request)
            )

            if not allowed:
                raise AppError(
                    429,
                    "rate_limited",
                    "Too many requests; slow down.",
                    headers={
                        "Retry-After": str(
                            max(1, int(retry_after) + 1)
                        )
                    },
                )

        return dependency

    # ---------------------------------------------------------
    # Middleware
    # ---------------------------------------------------------

    @app.middleware("http")
    async def base_middleware(request: Request, call_next):
        rid = uuid.uuid4().hex[:12]
        request.state.request_id = rid

        started = time.perf_counter()
        path = request.url.path

        # Health and readiness endpoints remain public.
        is_public = path in PUBLIC_PATHS

        # API key authentication.
        if settings.api_key and not is_public:
            authorization = request.headers.get(
                "authorization", ""
            )

            bearer = ""

            if authorization.lower().startswith("bearer "):
                bearer = authorization[7:].strip()

            candidate = (
                bearer
                or request.headers.get("x-api-key", "")
            )

            if not hmac.compare_digest(
                candidate,
                settings.api_key,
            ):
                response = JSONResponse(
                    AppError(
                        401,
                        "unauthorized",
                        "Valid API authentication is required.",
                    ).body(rid),
                    status_code=401,
                    headers={
                        "WWW-Authenticate": "Bearer",
                    },
                )

                response.headers["X-Request-ID"] = rid
                response.headers[
                    "X-Content-Type-Options"
                ] = "nosniff"
                response.headers["Cache-Control"] = "no-store"

                log_event(
                    logger,
                    "request",
                    request_id=rid,
                    method=request.method,
                    route="protected",
                    status=401,
                    latency_ms=round(
                        (time.perf_counter() - started) * 1000,
                        1,
                    ),
                )

                return response

        # Enforce declared body size for non-batch requests.
        # Batch requests have their own streaming size enforcement.
        content_length = request.headers.get(
            "content-length", ""
        )

        is_batch = path == f"{API}/predict/batch"

        if (
            request.method in {"POST", "PUT", "PATCH"}
            and not is_batch
            and content_length.isdigit()
            and int(content_length) > settings.max_json_bytes
        ):
            response = JSONResponse(
                AppError(
                    413,
                    "payload_too_large",
                    "Request body too large.",
                ).body(rid),
                status_code=413,
            )

        else:
            response = await call_next(request)

        response.headers["X-Request-ID"] = rid
        response.headers["X-Content-Type-Options"] = "nosniff"
        response.headers["Cache-Control"] = "no-store"

        route = request.scope.get("route")

        log_event(
            logger,
            "request",
            request_id=rid,
            method=request.method,
            route=route.path if route else "unmatched",
            status=response.status_code,
            latency_ms=round(
                (time.perf_counter() - started) * 1000,
                1,
            ),
        )

        return response

    # ---------------------------------------------------------
    # Exception handlers
    # ---------------------------------------------------------

    @app.exception_handler(AppError)
    async def handle_app_error(
        request: Request,
        exc: AppError,
    ):
        return JSONResponse(
            exc.body(request_id(request)),
            status_code=exc.status,
            headers=exc.headers,
        )

    @app.exception_handler(RequestValidationError)
    async def handle_validation_error(
        request: Request,
        exc: RequestValidationError,
    ):
        # Do not include submitted values in the response.
        details = [
            {
                "loc": [
                    str(item)
                    for item in error.get("loc", [])
                ],
                "reason": error.get("msg", "invalid"),
                "type": error.get("type", ""),
            }
            for error in exc.errors()
        ]

        error = AppError(
            422,
            "validation_error",
            "Request validation failed.",
            details=details,
        )

        return JSONResponse(
            error.body(request_id(request)),
            status_code=422,
        )

    @app.exception_handler(StarletteHTTPException)
    async def handle_http_error(
        request: Request,
        exc: StarletteHTTPException,
    ):
        code = {
            404: "not_found",
            405: "method_not_allowed",
        }.get(exc.status_code, "http_error")

        message = {
            404: "Resource not found.",
            405: "Method not allowed.",
        }.get(exc.status_code, "Request failed.")

        error = AppError(
            exc.status_code,
            code,
            message,
        )

        return JSONResponse(
            error.body(request_id(request)),
            status_code=exc.status_code,
            headers=exc.headers,
        )

    @app.exception_handler(Exception)
    async def handle_unexpected_error(
        request: Request,
        exc: Exception,
    ):
        log_event(
            logger,
            "unhandled_error",
            logging.ERROR,
            request_id=request_id(request),
            exc_type=type(exc).__name__,
        )

        error = AppError(
            500,
            "internal_error",
            "Internal server error.",
        )

        return JSONResponse(
            error.body(request_id(request)),
            status_code=500,
        )

    # ---------------------------------------------------------

    @app.post(
        f"{API}/predict",
        tags=["prediction"],
        response_model=PredictionOut,
        dependencies=[Depends(limit("predict"))],
    )
    def predict(tx: TransactionIn):
        payload = tx.model_dump(exclude_none=True)

        return service.predict_one(payload)

    # ---------------------------------------------------------
    # Batch CSV prediction
    # ---------------------------------------------------------

    @app.post(
        f"{API}/predict/batch",
        tags=["prediction"],
        response_model=BatchOut,
        dependencies=[Depends(limit("batch"))],
    )
    async def predict_batch(request: Request):
        """Accept raw CSV data without writing uploaded data to disk."""

        content_type = (
            request.headers.get("content-type", "")
            .split(";")[0]
            .strip()
            .lower()
        )

        if content_type not in CSV_TYPES:
            raise AppError(
                415,
                "unsupported_media_type",
                "Send the CSV as the raw body with Content-Type: text/csv.",
            )

        content_length = request.headers.get(
            "content-length", ""
        )

        if (
            content_length.isdigit()
            and int(content_length) > settings.max_upload_bytes
        ):
            raise AppError(
                413,
                "file_too_large",
                f"File exceeds the {settings.max_upload_bytes} byte limit.",
            )

        chunks = []
        total_bytes = 0

        async for chunk in request.stream():
            total_bytes += len(chunk)

            if total_bytes > settings.max_upload_bytes:
                raise AppError(
                    413,
                    "file_too_large",
                    f"File exceeds the {settings.max_upload_bytes} byte limit.",
                )

            chunks.append(chunk)

        csv_content = b"".join(chunks)

        return await run_in_threadpool(
            service.predict_batch,
            csv_content,
        )

    # ---------------------------------------------------------
    # Transaction listing and details
    # ---------------------------------------------------------

    @app.get(

"""Configuration from environment variables only.

No secrets are required or logged.
"""
from __future__ import annotations

import os
from dataclasses import dataclass
from pathlib import Path
from urllib.parse import unquote

from backend.app.errors import ConfigError


ROOT = Path(__file__).resolve().parents[2]

ALLOWED_ENVIRONMENTS = {
    "development",
    "test",
    "production",
}

ALLOWED_LOG_LEVELS = {
    "DEBUG",
    "INFO",
    "WARNING",
    "ERROR",
    "CRITICAL",
}


def _int(
    env,
    name: str,
    default: int,
    lo: int,
    hi: int,
) -> int:
    raw = env.get(name)

    if raw is None or raw == "":
        return default

    try:
        value = int(raw)
    except (TypeError, ValueError):
        raise ConfigError(
            f"{name} must be an integer"
        ) from None

    if not lo <= value <= hi:
        raise ConfigError(
            f"{name} must be between {lo} and {hi}"
        )

    return value


def _bool(
    env,
    name: str,
    default: bool = False,
) -> bool:
    """Parse an explicit Boolean environment variable."""
    raw = env.get(name)

    if raw is None or raw == "":
        return default

    value = str(raw).strip().lower()

    if value in ("true", "1", "yes", "on"):
        return True

    if value in ("false", "0", "no", "off"):
        return False

    raise ConfigError(
        f"{name} must be a Boolean value"
    )


def _path(
    value: str | Path,
    name: str,
) -> Path:
    """Normalize a configured filesystem path."""
    try:
        result = Path(value).expanduser()
    except (TypeError, ValueError, OSError):
        raise ConfigError(
            f"{name} must be a valid filesystem path"
        ) from None

    if not str(result):
        raise ConfigError(
            f"{name} must not be empty"
        )

    return result


def parse_sqlite_url(url: str) -> Path:
    """Parse a SQLite-only database URL.

    Supported examples:
        sqlite:///./fraud_platform.db
        sqlite:////absolute/path/fraud_platform.db
        sqlite:///C:/Projects/fraud_platform.db

    Relative paths remain relative to the application working directory.
    """
    prefix = "sqlite:///"

    if (
        not isinstance(url, str)
        or not url.startswith(prefix)
        or len(url) == len(prefix)
    ):
        raise ConfigError(
            "DATABASE_URL must look like sqlite:///./file.db (SQLite only)"
        )

    raw_path = unquote(url[len(prefix):])

    if not raw_path or "\x00" in raw_path:
        raise ConfigError(
            "DATABASE_URL must contain a valid SQLite file path"
        )

    if raw_path in (":memory:", "./:memory:"):
        return Path(":memory:")

    try:
        return Path(raw_path).expanduser()
    except (TypeError, ValueError, OSError):
        raise ConfigError(
            "DATABASE_URL must contain a valid SQLite file path"
        ) from None


@dataclass(frozen=True)
class Settings:
    app_env: str = "development"

    database_path: Path = Path("fraud_platform.db")

    model_dir: Path | None = None

    cors_origins: tuple = (
        "http://localhost:5173",
        "http://127.0.0.1:5173",
    )

    max_upload_bytes: int = 5_000_000
    max_csv_rows: int = 5_000
    max_json_bytes: int = 16_384
    max_page_size: int = 200

    rate_predict_per_min: int = 60
    rate_batch_per_min: int = 6
    rate_read_per_min: int = 240

    trust_proxy: bool = False

    log_level: str = "INFO"

    artifact_root: Path = ROOT / "ml" / "artifacts"

    outcomes_dir: Path = ROOT / "streaming_outcomes"

    min_monitor_rows: int = 100

    api_key: str | None = None

    @property
    def is_production(self) -> bool:
        return self.app_env == "production"

    @classmethod
    def from_env(cls, env=None) -> "Settings":
        """Build validated settings from environment variables."""
        env = os.environ if env is None else env

        # Environment
        app_env = str(
            env.get("APP_ENV", "development")
        ).strip().lower()

        if app_env not in ALLOWED_ENVIRONMENTS:
            raise ConfigError(
                "APP_ENV must be development, test or production"
            )

        defaults = cls()

        # CORS
        raw_origins = env.get(
            "CORS_ALLOW_ORIGINS",
            "",
        )

        origins = tuple(
            origin.strip().rstrip("/")
            for origin in raw_origins.split(",")
            if origin.strip()
        )

        if "*" in origins:
            raise ConfigError(
                "CORS_ALLOW_ORIGINS must list explicit origins; "
                "'*' is not allowed"
            )

        if any(
            not origin.startswith(("http://", "https://"))
            for origin in origins
        ):
            raise ConfigError(
                "CORS_ALLOW_ORIGINS must contain valid HTTP(S) origins"
            )

        # Paths
        database_url = env.get(
            "DATABASE_URL",
            "sqlite:///./fraud_platform.db",
        )

        database_path = parse_sqlite_url(database_url)

        model_dir_value = str(
            env.get("MODEL_ARTIFACT_DIR", "")
        ).strip()

        model_dir = (
            _path(model_dir_value, "MODEL_ARTIFACT_DIR")
            if model_dir_value
            else None
        )

        artifact_root = _path(
            env.get("ARTIFACT_ROOT") or defaults.artifact_root,
            "ARTIFACT_ROOT",
        )

        outcomes_dir = _path(
            env.get("STREAM_OUTCOMES_DIR") or defaults.outcomes_dir,
            "STREAM_OUTCOMES_DIR",
        )

        # Logging
        log_level = str(
            env.get("LOG_LEVEL", defaults.log_level)
        ).strip().upper()

        if log_level not in ALLOWED_LOG_LEVELS:
            raise ConfigError(
                "LOG_LEVEL must be DEBUG, INFO, WARNING, ERROR or CRITICAL"
            )

        # Proxy configuration
        trust_proxy = _bool(
            env,
            "TRUST_PROXY",
            defaults.trust_proxy,
        )

        # API key
        api_key = str(
            env.get("API_KEY", "")
        ).strip() or None

        if app_env == "production" and not api_key:
            raise ConfigError(
                "API_KEY must be configured in production"
            )

        return cls(
            app_env=app_env,

            database_path=database_path,

            model_dir=model_dir,

            cors_origins=origins or defaults.cors_origins,

            max_upload_bytes=_int(
                env,
                "MAX_UPLOAD_BYTES",
                defaults.max_upload_bytes,
                1_000,
                50_000_000,
            ),

            max_csv_rows=_int(
                env,
                "MAX_CSV_ROWS",
                defaults.max_csv_rows,
                1,
                100_000,
            ),

            max_json_bytes=_int(
                env,
                "MAX_JSON_BYTES",
                defaults.max_json_bytes,
                256,
                1_000_000,
            ),

            max_page_size=_int(
                env,
                "MAX_PAGE_SIZE",
                defaults.max_page_size,
                1,
                1_000,
            ),

            rate_predict_per_min=_int(
                env,
                "RATE_LIMIT_PREDICT_PER_MIN",
                defaults.rate_predict_per_min,
                1,
                100_000,
            ),

            rate_batch_per_min=_int(
                env,
                "RATE_LIMIT_BATCH_PER_MIN",
                defaults.rate_batch_per_min,
                1,
                10_000,
            ),

            rate_read_per_min=_int(
                env,
                "RATE_LIMIT_READ_PER_MIN",
                defaults.rate_read_per_min,
                1,
                100_000,
            ),

            trust_proxy=trust_proxy,

            log_level=log_level,

            artifact_root=artifact_root,

            outcomes_dir=outcomes_dir,

            min_monitor_rows=_int(
                env,
                "MIN_MONITOR_ROWS",
                defaults.min_monitor_rows,
                10,
                1_000_000,
            ),

            api_key=api_key,
        )

    "rejected": (),
    "cancelled": (),
}

SORT_COLUMNS = {
    "created_utc": "t.created_utc",
    "id": "t.id",
    "score": "t.score",
    "amount": "t.amount",
    "step": "t.step",
}

BALANCE_COLUMNS = (
    "oldbalance_orig",
    "newbalance_orig",
    "oldbalance_dest",
    "newbalance_dest",
)

MONITORING_COLUMNS = [
    "step",
    "type",
    "amount",
    "nameOrig",
    "nameDest",
    "score",
    "oldbalanceOrg",
    "newbalanceOrig",
    "oldbalanceDest",
    "newbalanceDest",
]


SCHEMA = """
CREATE TABLE IF NOT EXISTS transactions (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    created_utc TEXT NOT NULL,
    source TEXT NOT NULL CHECK (source IN ('single','batch')),
    batch_id TEXT,
    step INTEGER NOT NULL,
    type TEXT NOT NULL,
    amount REAL NOT NULL,
    name_orig TEXT NOT NULL,
    name_dest TEXT NOT NULL,
    oldbalance_orig REAL,
    newbalance_orig REAL,
    oldbalance_dest REAL,
    newbalance_dest REAL,
    score REAL NOT NULL,
    threshold REAL NOT NULL,
    flagged INTEGER NOT NULL CHECK (flagged IN (0,1)),
    type_in_training_vocab INTEGER NOT NULL,
    model_kind TEXT NOT NULL,
    model_sha256 TEXT NOT NULL,
    model_data_label TEXT
);

CREATE INDEX IF NOT EXISTS ix_tx_orig_step
    ON transactions(name_orig, step);

CREATE INDEX IF NOT EXISTS ix_tx_created
    ON transactions(created_utc);

CREATE INDEX IF NOT EXISTS ix_tx_flagged
    ON transactions(flagged);

CREATE INDEX IF NOT EXISTS ix_tx_batch
    ON transactions(batch_id);

CREATE TABLE IF NOT EXISTS cases (
    transaction_id INTEGER PRIMARY KEY
        REFERENCES transactions(id) ON DELETE CASCADE,
    status TEXT NOT NULL,
    note TEXT,
    updated_utc TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS case_events (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    transaction_id INTEGER NOT NULL
        REFERENCES transactions(id) ON DELETE CASCADE,
    from_status TEXT,
    to_status TEXT NOT NULL,
    created_utc TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS rejections (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    created_utc TEXT NOT NULL,
    endpoint TEXT NOT NULL,
    code TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS retrain_requests (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    created_utc TEXT NOT NULL,
    updated_utc TEXT NOT NULL,
    status TEXT NOT NULL,
    reason TEXT NOT NULL,
    note TEXT,
    candidate TEXT,
    evaluation_json TEXT
);

CREATE TABLE IF NOT EXISTS retrain_events (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    request_id INTEGER NOT NULL
        REFERENCES retrain_requests(id) ON DELETE CASCADE,
    from_status TEXT,
    to_status TEXT NOT NULL,
    note TEXT,
    created_utc TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS feedback (
    transaction_id INTEGER PRIMARY KEY
        REFERENCES transactions(id) ON DELETE CASCADE,
    label INTEGER NOT NULL CHECK (label IN (0,1)),
    comment TEXT,
    updated_utc TEXT NOT NULL
);

CREATE INDEX IF NOT EXISTS ix_case_events_transaction
    ON case_events(transaction_id);

CREATE INDEX IF NOT EXISTS ix_retrain_events_request
    ON retrain_events(request_id);
"""


def _now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def _positive_limit(value, field: str) -> int:
    if (
        not isinstance(value, int)
        or isinstance(value, bool)
        or value < 1
    ):
        raise ValueError(f"{field} must be a positive integer.")
    return value


class Repository:
    def __init__(self, path: Path):
        self.path = Path(path)

    @contextmanager
    def _conn(self, write: bool = False):
        """Open a connection and manage one explicit transaction."""
        conn = sqlite3.connect(
            self.path,
            timeout=10.0,
            isolation_level=None,
        )
        conn.row_factory = sqlite3.Row
        conn.execute("PRAGMA foreign_keys=ON")
        conn.execute("PRAGMA busy_timeout=10000")

        try:
            conn.execute(
                "BEGIN IMMEDIATE" if write else "BEGIN"
            )
            yield conn
            conn.execute("COMMIT")

        except BaseException:
            try:
                conn.execute("ROLLBACK")
            except sqlite3.Error:
                pass
            raise

        finally:
            conn.close()

    def init_schema(self) -> None:
        """Create tables and safely migrate older transaction schemas."""
        self.path.parent.mkdir(
            parents=True,
            exist_ok=True,
        )

        conn = sqlite3.connect(
            self.path,
            timeout=10.0,
        )

        try:
            conn.execute("PRAGMA foreign_keys=ON")
            conn.executescript(SCHEMA)

            existing = {
                row[1]
                for row in conn.execute(
                    "PRAGMA table_info(transactions)"
                )
            }

            for column in BALANCE_COLUMNS:
                if column not in existing:
                    # Column names come exclusively from BALANCE_COLUMNS.
                    conn.execute(
                        f"ALTER TABLE transactions ADD COLUMN {column} REAL"
                    )

            conn.commit()

        except BaseException:
            conn.rollback()
            raise

        finally:
            conn.close()

    def ping(self) -> bool:
        try:
            with self._conn() as conn:
                conn.execute(
                    "SELECT 1 FROM transactions LIMIT 1"
                ).fetchall()
            return True
        except sqlite3.Error:
            return False

    def insert_predictions(
        self,
        rows: list[dict],
        source: str,
        batch_id: str | None,
        model: dict,
    ) -> list[int]:
        if source not in ("single", "batch"):
            raise AppError(
                422,
                "validation_error",
                "Invalid prediction source.",
            )

        if not isinstance(rows, list) or not rows:
            raise AppError(
                422,
                "validation_error",
                "Prediction rows cannot be empty.",
            )

        if not isinstance(model, dict):
            raise AppError(
                422,
                "validation_error",
                "Invalid model metadata.",
            )

        required_model_fields = ("kind", "sha256")

        if any(not model.get(key) for key in required_model_fields):
            raise AppError(
                422,
                "validation_error",
                "Model metadata is incomplete.",
            )

        ids: list[int] = []
        now = _now()

        with self._conn(write=True) as conn:
            for row in rows:
                try:
                    step = int(row["step"])
                    amount = float(row["amount"])
                    score = float(row["score"])
                    threshold = float(row["threshold"])
                    flagged = bool(row["flagged"])
                    type_known = bool(row["type_known"])

                    if step < 0 or amount < 0:
                        raise ValueError("Invalid transaction values.")

                    if (
                        not 0.0 <= score <= 1.0
                        or not 0.0 <= threshold <= 1.0
                    ):
                        raise ValueError("Invalid score or threshold.")

                    transaction_type = str(row["type"])
                    name_orig = str(row["nameOrig"])
                    name_dest = str(row["nameDest"])

                    if not transaction_type or not name_orig or not name_dest:
                        raise ValueError("Transaction identifiers are empty.")

                    balances = [
                        (
                            None
                            if row.get(key) is None
                            else float(row[key])
                        )
                        for key in (
                            "oldbalanceOrg",
                            "newbalanceOrig",
                            "oldbalanceDest",
                            "newbalanceDest",
                        )
                    ]

                except (KeyError, TypeError, ValueError, OverflowError) as exc:
                    raise AppError(
                        422,
                        "validation_error",
                        "Invalid prediction row.",
                    ) from exc

                if any(
                    value is not None and (
                        not pd.notna(value) or value < 0
                    )
                    for value in balances
                ):
                    raise AppError(
                        422,
                        "validation_error",
                        "Invalid balance values.",
                    )

                cursor = conn.execute(
                    """
                    INSERT INTO transactions (
                        created_utc, source, batch_id,
                        step, type, amount, name_orig, name_dest,
                        score, threshold, flagged,
                        type_in_training_vocab,
                        model_kind, model_sha256, model_data_label,
                        oldbalance_orig, newbalance_orig,
                        oldbalance_dest, newbalance_dest
                    )
                    VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?,
                            ?, ?, ?, ?)
                    """,
                    (
                        now,
                        source,
                        batch_id,
                        step,
                        transaction_type,
                        amount,
                        name_orig,
                        name_dest,
                        score,
                        threshold,
                        int(flagged),
                        int(type_known),
                        model["kind"],
                        model["sha256"],
                        model.get("data_label"),
                        *balances,
                    ),
                )

                transaction_id = int(cursor.lastrowid)
                ids.append(transaction_id)

                if flagged:
                    conn.execute(
                        """
                        INSERT INTO cases (
                            transaction_id, status, note, updated_utc
                        )
                        VALUES (?, ?, ?, ?)
                        """,
                        (transaction_id, "open", None, now),
                    )

                    conn.execute(
                        """
                        INSERT INTO case_events (
                            transaction_id, from_status,
                            to_status, created_utc
                        )
                        VALUES (?, ?, ?, ?)
                        """,
                        (transaction_id, None, "open", now),
                    )

        return ids

    def _require_tx(self, conn, tid: int) -> None:
        if conn.execute(
            "SELECT 1 FROM transactions WHERE id=?",
            (tid,),
        ).fetchone() is None:
            raise AppError(
                404,
                "not_found",
                "Transaction not found.",
            )

    def update_case(
        self,
        tid: int,
        status: str,
        note: str | None,
    ) -> dict:
        if status not in CASE_STATUSES:
            raise AppError(
                422,
                "validation_error",
                "Invalid case status.",
                details={"allowed": list(CASE_STATUSES)},
            )

        now = _now()

        with self._conn(write=True) as conn:
            self._require_tx(conn, tid)

            previous = conn.execute(
                "SELECT status FROM cases WHERE transaction_id=?",
                (tid,),
            ).fetchone()

            if previous is None:
                conn.execute(
                    """
                    INSERT INTO cases (
                        transaction_id, status, note, updated_utc
                    )
                    VALUES (?, ?, ?, ?)
                    """,
                    (tid, status, note, now),
                )
            else:
                conn.execute(
                    """
                    UPDATE cases
                    SET status=?, note=?, updated_utc=?
                    WHERE transaction_id=?
                    """,
                    (status, note, now, tid),
                )

            conn.execute(
                """
                INSERT INTO case_events (
                    transaction_id, from_status, to_status, created_utc
                )
                VALUES (?, ?, ?, ?)
                """,
                (
                    tid,
                    previous["status"] if previous else None,
                    status,
                    now,
                ),
            )

        return {
            "transaction_id": tid,
            "status": status,
            "note": note,
            "updated_utc": now,
        }

    def set_feedback(
        self,
        tid: int,
        label: int,
        comment: str | None,
    ) -> dict:
        if (
            not isinstance(label, int)
            or isinstance(label, bool)
            or label not in (0, 1)
        ):
            raise AppError(
                422,
                "validation_error",
                "label must be integer 0 or 1.",
            )

        now = _now()

        with self._conn(write=True) as conn:
            self._require_tx(conn, tid)

            conn.execute(
                """
                INSERT INTO feedback (
                    transaction_id, label, comment, updated_utc
                )
                VALUES (?, ?, ?, ?)
                ON CONFLICT(transaction_id)
                DO UPDATE SET
                    label=excluded.label,
                    comment=excluded.comment,
                    updated_utc=excluded.updated_utc
                """,
                (tid, label, comment, now),
            )

        return {
            "transaction_id": tid,
            "label": label,
            "comment": comment,
            "updated_utc": now,
        }

    # ------------------------------------------------------------------
    # Reads
    # ------------------------------------------------------------------

    @staticmethod
    def _row(row: sqlite3.Row) -> dict:
        result = dict(row)
        result["flagged"] = bool(result["flagged"])
        result["type_in_training_vocab"] = bool(
            result["type_in_training_vocab"]
        )
        return result

    _SELECT = (
        "SELECT t.*, "
        "c.status AS case_status, "
        "c.note AS case_note, "
        "f.label AS feedback_label "
        "FROM transactions t "
        "LEFT JOIN cases c ON c.transaction_id=t.id "
        "LEFT JOIN feedback f ON f.transaction_id=t.id"
    )

    def get_transaction(self, tid: int) -> dict:
        with self._conn() as conn:
            row = conn.execute(
                self._SELECT + " WHERE t.id=?",
                (tid,),
            ).fetchone()

            if row is None:
                raise AppError(
                    404,
                    "not_found",
                    "Transaction not found.",
                )

            events = [
                dict(event)
                for event in conn.execute(
                    """
                    SELECT from_status, to_status, created_utc
                    FROM case_events
                    WHERE transaction_id=?
                    ORDER BY id
                    """,
                    (tid,),
                )
            ]

        result = self._row(row)
        result["case_history"] = events
        return result

    def list_transactions(
        self,
        *,
        page: int,
        page_size: int,
        sort: str = "id",
        order: str = "desc",
        flagged: bool | None = None,
        type_: str | None = None,
        min_score: float | None = None,
        max_score: float | None = None,
        source: str | None = None,
        case_status: str | None = None,
        batch_id: str | None = None,
    ) -> dict:
        if (
            not isinstance(page, int)
            or isinstance(page, bool)
            or page < 1
            or not isinstance(page_size, int)
            or isinstance(page_size, bool)
            or page_size < 1
        ):
            raise AppError(
                422,
                "validation_error",
                "Invalid pagination.",
            )

        if sort not in SORT_COLUMNS or order not in ("asc", "desc"):
            raise AppError(
                422,
                "validation_error",
                "Invalid sort or order.",
                details={
                    "sort_allowed": sorted(SORT_COLUMNS),
                    "order_allowed": ["asc", "desc"],
                },
            )

        if flagged is not None and not isinstance(flagged, bool):
            raise AppError(
                422,
                "validation_error",
                "flagged must be a Boolean.",
            )

        if source not in (None, "single", "batch"):
            raise AppError(
                422,
                "validation_error",
                "Invalid source.",
            )

        if case_status is not None and case_status not in CASE_STATUSES:
            raise AppError(
                422,
                "validation_error",
                "Invalid case status.",
            )

        for value, field in (
            (min_score, "min_score"),
            (max_score, "max_score"),
        ):
            if value is not None:
                try:
                    numeric = float(value)
                except (TypeError, ValueError):
                    raise AppError(
                        422,
                        "validation_error",
                        f"{field} must be numeric.",
                    ) from None

                if not 0.0 <= numeric <= 1.0:
                    raise AppError(
                        422,
                        "validation_error",
                        f"{field} must be between 0 and 1.",
                    )

        if (
            min_score is not None
            and max_score is not None
            and min_score > max_score
        ):
            raise AppError(
                422,
                "validation_error",
                "min_score cannot exceed max_score.",
            )

        conditions = []
        parameters = []

        filters = (
            ("t.flagged=?", None if flagged is None else int(flagged)),
            ("t.type=?", type_),
            ("t.score>=?", min_score),
            ("t.score<=?", max_score),
            ("t.source=?", source),
            ("c.status=?", case_status),
            ("t.batch_id=?", batch_id),
        )

        for condition, value in filters:
            if value is not None:
                conditions.append(condition)
                parameters.append(value)

        where = (
            " WHERE " + " AND ".join(conditions)
            if conditions
            else ""
        )

        base = (
            " FROM transactions t "
            "LEFT JOIN cases c ON c.transaction_id=t.id "
            + where
        )

        offset = (page - 1) * page_size

        with self._conn() as conn:
            total = conn.execute(
                "SELECT COUNT(*)" + base,
                parameters,
            ).fetchone()[0]

            rows = conn.execute(
                self._SELECT
                + where
                + f" ORDER BY {SORT_COLUMNS[sort]} {order.upper()}, "
                  "t.id DESC LIMIT ? OFFSET ?",
                [*parameters, page_size, offset],
            ).fetchall()

        return {
            "items": [self._row(row) for row in rows],
            "total": int(total),
            "page": page,
            "page_size": page_size,
        }

    def history_for(
        self,
        names: list[str],
        limit: int = 50_000,
    ) -> list[dict]:
        """Return previously scored transactions for requested senders."""
        _positive_limit(limit, "limit")

        if not names:
            return []

        unique_names = list(dict.fromkeys(names))
        output: list[dict] = []

        with self._conn() as conn:
            for index in range(0, len(unique_names), 500):
                chunk = unique_names[index:index + 500]

                placeholders = ",".join("?" for _ in chunk)
                remaining = limit - len(output)

                if remaining <= 0:
                    break

                rows = conn.execute(
                    """
                    SELECT
                        step,
                        type,
                        amount,
                        name_orig AS nameOrig,
                        name_dest AS nameDest
                    FROM transactions
                    WHERE name_orig IN (
                    """
                    + placeholders
                    + """
                    )
                    ORDER BY id DESC
                    LIMIT ?
                    """,
                    [*chunk, remaining],
                )

                output.extend(
                    dict(row)
                    for row in rows
                )

        return output

    def summary(self) -> dict:
        with self._conn() as conn:
            total, flagged = conn.execute(
                """
                SELECT
                    COUNT(*),
                    COALESCE(SUM(flagged), 0)
                FROM transactions
                """
            ).fetchone()

            histogram = [0] * 10

            for row in conn.execute(
                """
                SELECT
                    MIN(CAST(score * 10 AS INTEGER), 9) AS bin,
                    COUNT(*) AS count
                FROM transactions
                GROUP BY bin
                """
            ):
                bin_index = max(
                    0,
                    min(9, row["bin"]),
                )
                histogram[bin_index] += row["count"]

            by_day = [
                dict(row)
                for row in conn.execute(
                    """
                    SELECT
                        substr(created_utc, 1, 10) AS day,
                        COUNT(*) AS transactions,
                        SUM(flagged) AS flagged
                    FROM transactions
                    GROUP BY day
                    ORDER BY day DESC
                    LIMIT 30
                    """
                )
            ][::-1]

            by_type = [
                dict(row)
                for row in conn.execute(
                    """
                    SELECT
                        type,
                        COUNT(*) AS transactions,
                        SUM(flagged) AS flagged
                    FROM transactions
                    GROUP BY type
                    ORDER BY transactions DESC
                    """
                )
            ]

            cases = {
                row["status"]: row["count"]
                for row in conn.execute(
                    """
                    SELECT status, COUNT(*) AS count
                    FROM cases
                    GROUP BY status
                    """
                )
            }

            feedback = {
                row["label"]: row["count"]
                for row in conn.execute(
                    """
                    SELECT label, COUNT(*) AS count
                    FROM feedback
                    GROUP BY label
                    """
                )
            }

        return {
            "total_transactions": int(total),
            "flagged_transactions": int(flagged),
            "score_histogram": [
                {
                    "bin_start": index / 10,
                    "bin_end": (index + 1) / 10,
                    "count": count,
                }
                for index, count in enumerate(histogram)
            ],
            "by_day": by_day,
            "by_type": by_type,
            "case_status_counts": {
                status: cases.get(status, 0)
                for status in CASE_STATUSES
            },
            "feedback": {
                "total": sum(feedback.values()),
                "labelled_fraud": feedback.get(1, 0),
                "labelled_not_fraud": feedback.get(0, 0),
            },
        }

    def record_rejection(
        self,
        endpoint: str,
        code: str,
    ) -> None:
        try:
            with self._conn(write=True) as conn:
                conn.execute(
                    """
                    INSERT INTO rejections (
                        created_utc, endpoint, code
                    )
                    VALUES (?, ?, ?)
                    """,
                    (_now(), endpoint[:40], code[:40]),
                )
        except sqlite3.Error:
            # Rejection monitoring must not break request handling.
            pass

    def rejection_counts(self) -> dict:
        with self._conn() as conn:
            return {
                row["code"]: row["count"]
                for row in conn.execute(
                    """
                    SELECT code, COUNT(*) AS count
                    FROM rejections
                    GROUP BY code
                    ORDER BY count DESC
                    """
                )
            }

    def monitoring_rows(
        self,
        limit: int = 5000,
    ) -> pd.DataFrame:
        _positive_limit(limit, "limit")

        with self._conn() as conn:
            rows = conn.execute(
                """
                SELECT
                    step,
                    type,
                    amount,
                    name_orig AS nameOrig,
                    name_dest AS nameDest,
                    score,
                    oldbalance_orig AS oldbalanceOrg,
                    newbalance_orig AS newbalanceOrig,
                    oldbalance_dest AS oldbalanceDest,
                    newbalance_dest AS newbalanceDest
                FROM (
                    SELECT *
                    FROM transactions
                    ORDER BY id DESC
                    LIMIT ?
                )
                ORDER BY id ASC
                """,
                (limit,),
            ).fetchall()

        dataframe = pd.DataFrame(
            [dict(row) for row in rows],
            columns=MONITORING_COLUMNS,
        )

        numeric_columns = (
            "step",
            "amount",
            "score",
            "oldbalanceOrg",
            "newbalanceOrig",
            "oldbalanceDest",
            "newbalanceDest",
        )

        for column in numeric_columns:
            dataframe[column] = pd.to_numeric(
                dataframe[column],
                errors="coerce",
            )

        return dataframe

    def labelled_rows(
        self,
        limit: int = 5000,
    ) -> pd.DataFrame:
        _positive_limit(limit, "limit")

        with self._conn() as conn:
            rows = conn.execute(
                """
                SELECT
                    t.score AS score,
                    t.threshold AS threshold,
                    f.label AS label
                FROM feedback f
                JOIN transactions t
                    ON t.id=f.transaction_id
                ORDER BY f.updated_utc DESC
                LIMIT ?
                """,
                (limit,),
            ).fetchall()

        dataframe = pd.DataFrame(
            [dict(row) for row in rows],
            columns=["score", "threshold", "label"],
        )

        for column in ("score", "threshold", "label"):
            dataframe[column] = pd.to_numeric(
                dataframe[column],
                errors="coerce",
            )

        return dataframe

    def create_retrain(
        self,
        reason: str,
        note: str | None,
    ) -> dict:
        now = _now()

        with self._conn(write=True) as conn:
            cursor = conn.execute(
                """
                INSERT INTO retrain_requests (
                    created_utc,
                    updated_utc,
                    status,
                    reason,
                    note
                )
                VALUES (?, ?, ?, ?, ?)
                """,
                (now, now, "requested", reason, note),
            )

            request_id = int(cursor.lastrowid)

            conn.execute(
                """
                INSERT INTO retrain_events (
                    request_id,
                    from_status,
                    to_status,
                    note,
                    created_utc
                )
                VALUES (?, ?, ?, ?, ?)
                """,
                (request_id, None, "requested", note, now),
            )

        return self.get_retrain(request_id)

    def get_retrain(self, rid: int) -> dict:
        with self._conn() as conn:
            row = conn.execute(
                """
                SELECT *
                FROM retrain_requests
                WHERE id=?
                """,
                (rid,),
            ).fetchone()

            if row is None:
                raise AppError(
                    404,
                    "not_found",
                    "Retraining request not found.",
                )

            events = [
                dict(event)
                for event in conn.execute(
                    """
                    SELECT
                        from_status,
                        to_status,
                        note,
                        created_utc
                    FROM retrain_events
                    WHERE request_id=?
                    ORDER BY id
                    """,
                    (rid,),
                )
            ]

        result = dict(row)

        try:
            evaluation_json = result.pop("evaluation_json")
            result["evaluation"] = (
                json.loads(evaluation_json)
                if evaluation_json
                else None
            )
        except (ValueError, TypeError):
            result["evaluation"] = None
            result["evaluation_error"] = "Stored evaluation JSON is invalid."

        result["events"] = events
        result["allowed_next"] = list(
            RETRAIN_NEXT.get(result["status"], ())
        )

        return result

    def list_retrain(self) -> list[dict]:
        with self._conn() as conn:
            ids = [
                row["id"]
                for row in conn.execute(
                    """
                    SELECT id
                    FROM retrain_requests
                    ORDER BY id DESC
                    LIMIT 100
                    """
                )
            ]

        return [
            self.get_retrain(request_id)
            for request_id in ids
        ]

    def transition_retrain(
        self,
        rid: int,
        to: str,
        note: str | None,
        candidate: str | None = None,
        evaluation: dict | None = None,
    ) -> dict:
        now = _now()

        with self._conn(write=True) as conn:
            row = conn.execute(
                """
                SELECT status
                FROM retrain_requests
                WHERE id=?
                """,
                (rid,),
            ).fetchone()

            if row is None:
                raise AppError(
                    404,
                    "not_found",
                    "Retraining request not found.",
                )

            current_status = row["status"]

            if to not in RETRAIN_NEXT.get(current_status, ()):
                raise AppError(
                    409,
                    "invalid_transition",
                    f"Cannot move from '{current_status}' to '{to}'.",
                    details={
                        "allowed": list(
                            RETRAIN_NEXT.get(current_status, ())
                        ),
                    },
                )

            try:
                evaluation_json = (
                    json.dumps(
                        evaluation,
                        allow_nan=False,
                    )
                    if evaluation is not None
                    else None
                )
            except (TypeError, ValueError):
                raise AppError(
                    422,
                    "validation_error",
                    "Evaluation data must be valid JSON.",
                ) from None

            conn.execute(
                """
                UPDATE retrain_requests
                SET
                    status=?,
                    updated_utc=?,
                    candidate=COALESCE(?, candidate),
                    evaluation_json=COALESCE(?, evaluation_json)
                WHERE id=?
                """,
                (
                    to,
                    now,
                    candidate,
                    evaluation_json,
                    rid,
                ),
            )

            conn.execute(
                """
                INSERT INTO retrain_events (
                    request_id,
                    from_status,
                    to_status,
                    note,
                    created_utc
                )
                VALUES (?, ?, ?, ?, ?)
                """,
                (
                    rid,
                    current_status,
                    to,
                    note,
                    now,
                ),
            )

        return self.get_retrain(rid)

"""Structured logging that excludes transaction data and sensitive values.

Only whitelisted scalar fields are allowed.
String values are sanitized to prevent log injection.
"""

from __future__ import annotations

import logging
import math
import re
from typing import Any


SAFE_FIELDS = {
    "request_id",
    "method",
    "route",
    "status",
    "latency_ms",
    "rows",
    "flagged",
    "model_state",
    "error_code",
    "exc_type",
    "batch_id",
    "count",
}

MAX_EVENT_LENGTH = 64
MAX_VALUE_LENGTH = 256

SAFE_LOG_LEVELS = {
    "DEBUG": logging.DEBUG,
    "INFO": logging.INFO,
    "WARNING": logging.WARNING,
    "ERROR": logging.ERROR,
    "CRITICAL": logging.CRITICAL,
}

SAFE_EVENT_PATTERN = re.compile(r"^[A-Za-z0-9_.:-]+$")
SAFE_STRING_PATTERN = re.compile(r"[\r\n\t\x00-\x1f\x7f]")


def _clean_string(value: str, max_length: int) -> str:
    """Sanitize a string before adding it to a log record."""

    value = SAFE_STRING_PATTERN.sub("_", value)
    value = value[:max_length]

    return value


def _safe_value(value: Any) -> str | int | float | bool | None:
    """Return a safe scalar value or None."""

    if isinstance(value, bool):
        return value

    if isinstance(value, int):
        return value

    if isinstance(value, float):
        if not math.isfinite(value):
            return None
        return value

    if isinstance(value, str):
        return _clean_string(value, MAX_VALUE_LENGTH)

    return None


def setup_logging(level: str = "INFO") -> logging.Logger:
    """Configure the application logger."""

    logger = logging.getLogger("fraudapp")

    normalized_level = (
        level.upper()
        if isinstance(level, str)
        else "INFO"
    )

    numeric_level = SAFE_LOG_LEVELS.get(
        normalized_level,
        logging.INFO,
    )

    if not logger.handlers:
        handler = logging.StreamHandler()

        handler.setFormatter(
            logging.Formatter(
                "%(asctime)s %(levelname)s %(message)s"
            )
        )

        logger.addHandler(handler)

    logger.setLevel(numeric_level)

    # Prevent duplicate records through the root logger.
    logger.propagate = False

    return logger


def log_event(
    logger: logging.Logger,
    event: str,
    level: int = logging.INFO,
    **fields: Any,
) -> None:
    """Write an event using only approved scalar fields."""

    if not isinstance(event, str):
        event = "unknown_event"

    event = _clean_string(event, MAX_EVENT_LENGTH)

    if not SAFE_EVENT_PATTERN.fullmatch(event):
        event = "invalid_event"

    safe_fields = {}

    for key, value in fields.items():
        if key not in SAFE_FIELDS:
            continue

        safe_value = _safe_value(value)

        if safe_value is None:
            continue

        safe_fields[key] = safe_value

    details = " ".join(
        f"{key}={value}"
        for key, value in sorted(safe_fields.items())
    )

    if not isinstance(level, int) or isinstance(level, bool):
        level = logging.INFO

    logger.log(
        level,
        "%s %s",
        event,
        details,
    )
